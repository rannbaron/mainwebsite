\noindent \textbf{\large Math 105L} \hfill \textbf{\large Lecture 4-3}\\[1pt]
{\textbf{\Large The Derivative at a Point: Secant Lines to Tangent Lines}}\\[-5pt]
\rule{\textwidth}{1.2pt}

\vspace{0.03in}

\textbf{\large 1. Review: Rates of Change \& Graphical Meaning}

\begin{itemize}[leftmargin=1.2em, topsep=1pt, itemsep=3pt]
  \item \textbf{Average Velocity / Rate of Change on $[a, a+h]$:}
  \[
    \text{Avg RoC} = \frac{f(a+h) - f(a)}{(a+h) - a} = \mathbf{\frac{f(a+h) - f(a)}{h}}
  \]
  \textbf{Graphical Meaning:} The slope of the \textbf{secant line} connecting the two distinct points $(a, f(a))$ and $(a+h, f(a+h))$.
  
  \item \textbf{Instantaneous Rate of Change at $x = a$:}
  \[
    \text{Inst RoC} = \mathbf{\lim_{h\to 0}\frac{f(a+h) - f(a)}{h}}
  \]
  \textbf{Graphical Meaning:} The slope of the \textbf{tangent line} to the curve at the single point $(a, f(a))$.
\end{itemize}

\vspace{0.03in}

\begin{tcolorbox}[colback=boxbg, colframe=boxframe, boxrule=0.8pt, arc=3pt, left=8pt, right=8pt, top=4pt, bottom=4pt]
\textbf{\large Formal Definition of the Derivative at a Point}
Given a function $f(x)$, we define its \textbf{derivative at the point $x = a$}, denoted $f'(a)$, by:
\[
  \mathbf{f'(a) = \lim_{h\to 0}\frac{f(a+h) - f(a)}{h}}
\]
provided this limit exists. If the limit exists, we say $f$ is \textbf{differentiable at $a$}.
\end{tcolorbox}

\vspace{0.04in}

\textbf{\large 2. Worked Classroom Example: Estimating $E'(7)$ for $E(t) = \frac{6}{t+1} + 4$}

Let $E(t)$ represent student enrollment (in thousands) in a program $t$ days after launch:
\[
  E(t) = \frac{6}{t+1} + 4
\]
We estimate the instantaneous rate of change at $t = 7$ days using difference quotients:
\[
  \text{Avg RoC on } [7, 7+h] = \frac{E(7+h) - E(7)}{h} \quad \text{where } E(7) = \frac{6}{7+1} + 4 = \frac{6}{8} + 4 = 4.75
\]

\textbf{Sample Computation with $h = 0.1$:}
\[
  E(7.1) = \frac{6}{7.1+1} + 4 = \frac{6}{8.1} + 4 \approx 4.74074
\]
\[
  \text{Avg RoC} = \frac{4.74074 - 4.75}{0.1} = \frac{-0.00926}{0.1} = \mathbf{-0.0926} \text{ thousand students/day}
\]

\textbf{Behavior as $h \to 0$ from Both Sides:}
\begin{center}
\begin{tabular}{|c|c||c|c|}
  \hline
  $h$ (days) & Avg RoC on $[7, 7+h]$ & $h$ (days) & Avg RoC on $[7, 7+h]$ \\
  \hline
  $-0.1$   & $-0.0949$ & $0.1$   & $-0.0926$ \\
  $-0.01$  & $-0.0939$ & $0.01$  & $-0.0936$ \\
  $-0.001$ & $-0.0938$ & $0.001$ & $-0.0937$ \\
  \hline
\end{tabular}
\end{center}

\begin{tcolorbox}[colback=chalkblue!6, colframe=chalkblue, boxrule=1pt, arc=3pt, left=8pt, right=8pt, top=4pt, bottom=4pt]
\textbf{Conclusion:}\\[1pt]
As $h \to 0$ from both the left and right, the difference quotients converge to:
\[
  \mathbf{E'(7) = -0.09375 \text{ thousand students/day} = -93.75 \text{ students/day}}
\]
The negative sign confirms that enrollment is dropping by approximately 94 students per day at $t = 7$.
\end{tcolorbox}
