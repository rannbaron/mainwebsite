\section*{1. Preliminaries: Connecting $f, f'$, and $f''$}
\begin{itemize}[leftmargin=1.5em]
  \item If $f'(x) > 0$ on an interval, then $f(x)$ is \textbf{\textcolor{chalkblue}{increasing}} on that interval.
  \item If $f'(x) < 0$ on an interval, then $f(x)$ is \textbf{\textcolor{chalkblue}{decreasing}} on that interval.
  \item If $f''(x) > 0$ on an interval, then $f(x)$ is \textbf{\textcolor{chalkblue}{concave up}} on that interval ($f'$ is increasing).
  \item If $f''(x) < 0$ on an interval, then $f(x)$ is \textbf{\textcolor{chalkblue}{concave down}} on that interval ($f'$ is decreasing).
\end{itemize}

\section*{2. The First Derivative Test and Critical Points}
Let $a$ be an interior point in the domain of $f$:
\begin{itemize}[leftmargin=1.5em]
  \item $a$ is called a \textbf{critical point} of $f$ if $f'(a) = 0$ or $f'(a)$ is \textbf{undefined}.
  \item $f$ has a \textbf{local maximum} at $a$ if $f'(x)$ changes sign from \textbf{positive to negative} as $x$ increases through $a$.
  \item $f$ has a \textbf{local minimum} at $a$ if $f'(x)$ changes sign from \textbf{negative to positive} as $x$ increases through $a$.
  \item If $f'(x)$ does not change sign across $a$, then $a$ is neither a local maximum nor a local minimum.
\end{itemize}

\section*{3. The Second Derivative Test}
Suppose $f'(a) = 0$ and $f''$ is continuous near $a$:
\begin{itemize}[leftmargin=1.5em]
  \item If $f''(a) < 0$, the graph is concave down at $a$, so $f$ has a \textbf{local maximum} at $x = a$.
  \item If $f''(a) > 0$, the graph is concave up at $a$, so $f$ has a \textbf{local minimum} at $x = a$.
  \item If $f''(a) = 0$, the Second Derivative Test is \textbf{inconclusive}; use the First Derivative Test.
\end{itemize}

\section*{4. Points of Inflection}
A point $p = (c, f(c))$ on the curve is an \textbf{inflection point} if:
\begin{enumerate}[leftmargin=1.5em]
  \item $f$ is continuous at $c$, and
  \item The concavity of $f$ changes sign at $c$ (i.e. $f''$ changes from positive to negative, or negative to positive).
\end{enumerate}
Note: $f''(c) = 0$ is a necessary candidate condition, but not sufficient on its own (concavity must change sign).
